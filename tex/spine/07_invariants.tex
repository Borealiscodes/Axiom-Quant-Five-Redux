\[
D_5 = \{ x \in \mathcal{M} \mid t(x) \in \mathbb{R}^{80} \text{ is smooth} \}.
\]

Altitude alignment:
\[
A9 \leftrightarrow D_5.

\[

\[
\Phi_9 : \mathbb{R}^{80} \to \mathbb{R}^5.

\subsection{Connectivity Invariant}

Define the connectivity invariant:
\[
C_{\text{conn}}(x)
= \sum_{i=1}^{80} |t_i(x)|.
\]

This measures global geometric connectivity.

Properties:
\[
C_{\text{conn}} \in C^\infty,
\qquad
C_{\text{conn}} \ge 0.
\]

\subsection{Ternary Quantization Invariant}

Define ternary quantization:
\[
Q_{\text{tern}}(x)
= \sum_{i=1}^{80} \operatorname{sgn}(t_i(x)),
\]
where:
\[
\operatorname{sgn}(u) =
\begin{cases}
+1 & u > 0, \\
0 & u = 0, \\
-1 & u < 0.
\end{cases}
\]

This measures directional topology.

\subsection{Convexity Invariant}

Define convexity:
\[
V_{\text{conv}}(x)
= \sum_{i=1}^{80} t_i(x)^2.
\]

This measures geometric convexity of the topology-modulated vector.

\subsection{Canonical Form Invariant}

Define canonical form:
\[
F_{\text{canon}}(x)
= \left\| t(x) - \frac{1}{80}\sum_{i=1}^{80} t_i(x) \right\|.
\]

This measures deviation from canonical geometric form.

\subsection{Entropy Shift Invariant}

Define entropy shift:
\[
S_{\text{entropy}}(x)
= -\sum_{i=1}^{80}
\left(
p_i(x) \log p_i(x)
\right),
\]
where:
\[
p_i(x) = \frac{|t_i(x)|}{\sum_{j=1}^{80} |t_j(x)|}.
\]

This measures geometric entropy of the topology-modulated vector.

Thus:
\[
\Phi_9 \in C^\infty(\mathcal{M}, \mathbb{R}^5)
\quad\text{except at } t_i = 0.
